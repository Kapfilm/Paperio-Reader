if(NOT APPLE)
  find_package(OpenSSL REQUIRED)
endif()

foreach(target IN ITEMS FirmwareImageTest FirmwareImageClassicTest)
  add_executable(${target} FirmwareImageTest.cpp)
  target_include_directories(${target} BEFORE PRIVATE shims)
  target_link_libraries(${target} PRIVATE crosspoint_test_common GTest::gtest_main)
  if(APPLE)
    target_compile_options(${target} PRIVATE -Wno-deprecated-declarations)
  else()
    target_link_libraries(${target} PRIVATE OpenSSL::Crypto)
  endif()
endforeach()

target_compile_definitions(FirmwareImageClassicTest PRIVATE FREEINK_DEVICE_X4CLASSIC=1)
gtest_discover_tests(FirmwareImageTest)
